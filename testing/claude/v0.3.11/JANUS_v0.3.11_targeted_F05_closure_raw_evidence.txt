// Independent build-info generator (probe tool, not part of the JANUS repo).
// Usage: node gen_buildinfo.mjs <repoRoot> <commit>  -> writes <repoRoot>/build-info.json
import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";
import {spawnSync} from "node:child_process";
const [root, commit] = process.argv.slice(2);
const git = (args, enc = "utf8") => {
  const r = spawnSync("git", ["-C", root, ...args], {encoding: enc, maxBuffer: 1 << 28});
  if (r.status !== 0) throw new Error(String(r.stderr));
  return r.stdout;
};
const full = git(["rev-parse", commit + "^{commit}"]).trim();
const paths = git(["ls-tree", "-r", "--name-only", "-z", full]).split("\0").filter(Boolean)
  .filter(p => p !== "build-info.json").sort();
const files = {};
for (const p of paths) files[p] = crypto.createHash("sha256").update(git(["show", `${full}:${p}`], "buffer")).digest("hex");
const out = {version: "0.3.11", build_id: "janus-governance-challenge-harness-v0.3.11", code_commit: full, files};
fs.writeFileSync(path.join(root, "build-info.json"), JSON.stringify(out, null, 2) + "\n");
console.log("generated build-info for", full, Object.keys(files).length, "files");

    e = dict(os.environ); e.update(env or {})
    r = subprocess.run(["bash", "-c", cmd], cwd=cwd, env=e, capture_output=True, text=True)
    return r.returncode, (r.stdout + r.stderr).strip()
